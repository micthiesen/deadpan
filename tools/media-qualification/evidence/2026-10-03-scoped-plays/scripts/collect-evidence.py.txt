def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def compress(source, target):
    target.write_bytes(gzip.compress(source.read_bytes(), mtime=0))

def run(*args):
    return subprocess.check_output(args, text=True).strip()

def read(name):
    return json.loads((scratch / name).read_text())

def source_diff(first, last):
    a = read('source-' + first + '.json')
    b = read('source-' + last + '.json')
    return sorted(k for k in a.keys() | b.keys() if a.get(k) != b.get(k))

    source = scratch / name / 'report.json'

    scenario = next(s for s in report['scenarios'] if s['name'] != 'kestrel-shortcuts')
