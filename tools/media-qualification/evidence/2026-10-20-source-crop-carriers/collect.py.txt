from pathlib import Path
import gzip, hashlib, json, re, shutil
root = Path('/Users/michael/Code/deadpan')
scratch = Path('/tmp/deadpan-source-crop-carriers-2026-10-20')
dest = root / 'tools/media-qualification/evidence/2026-10-20-source-crop-carriers'
def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
check = (scratch / 'check.log').read_text()
assert 'Summary [' in check and re.search(r'(\d+) tests run: \1 passed', check), check[-2000:]
rows = json.loads((scratch / 'export-results.json').read_text())
assert len(rows) == 2
for row in rows:
    report = row['report']
    assert not report['failures']
    assert report['summary']['pictures_checked'] == (19 if row['name'] == 'shrink' else 34)
    assert report['summary']['pictures_failed'] == 0
    assert report['summary']['nonzero_offsets'] == 0
    for audio in report['audio']:
        assert audio['passed'] and not audio['flags']
        assert audio['uncovered_samples'] == 0
        assert audio['offset_status'] == 'verified_zero'
        assert audio['measured_offset_samples'] == 0
        for key in ['low_snr_blocks', 'level_blocks', 'sound_in_silence_blocks']:
            assert audio['blocks'][key] == 0
    movie = Path(row['movie'])
    row['retained_movie'] = 'clocked-speed-' + row['name'] + '.mp4'
    row['emitted_sha256'] = sha(movie)
    shutil.copy2(movie, scratch / row['retained_movie'])
(scratch / 'export-results.json').write_text(json.dumps(rows, indent=2) + '\n')
source_files = [
    'Cargo.lock',
    'crates/deadpan-core/src/insert_time/retimed/speed_update.rs',
    'crates/deadpan-plan/src/audio_bound.rs',
    'crates/deadpan-audio/src/stages.rs',
    'crates/deadpan-core/tests/insert_time/retimed/speed_update.rs',
    'crates/deadpan-core/tests/insert_time/retimed/speed_update/source_crop.rs',
    'crates/deadpan-audio/tests/stages/clocked_speed_update.rs',
    'crates/deadpan-audio/tests/stages/clocked_speed_update/source_crop_pcm.rs',
]
manifest = {'base': 'd0f675a5', 'sha256': {p:sha(root/p) for p in source_files}}
assert manifest['sha256']['Cargo.lock'] == '1e39d18ae4de6c7b898edad39eb7104235fb3ffe3af84521ca94df2b6c25b71b'
manifest['release_binaries'] = json.loads((scratch/'export-binaries.json').read_text())
for name,digest in manifest['release_binaries'].items():
    assert sha(root/name) == digest, name
(scratch/'source-identities.json').write_text(json.dumps(manifest,indent=2)+'\n')
dest.mkdir(parents=True,exist_ok=True)
for old in dest.rglob('*.patch.txt'):
    old.unlink()
keep = {'.txt','.md','.json','.log','.diff','.patch','.sh','.py','.rs','.exit','.mp4'}
exclude = {'next-repeat-family','support-boundaries','reconstructed-old'}
for path in sorted(scratch.rglob('*')):
    relative = path.relative_to(scratch)
    if (not path.is_file() or path.suffix not in keep or relative.parts[0] == 'next-repeat-family'
        or path.name in {'collection-summary.json'}):
        continue
    out=dest/relative
    out.parent.mkdir(parents=True,exist_ok=True)
    if path.suffix in {'.log','.json','.diff','.patch'}:
        out.with_name(out.name+'.gz').write_bytes(gzip.compress(path.read_bytes(),mtime=0))
    elif path.suffix in {'.sh','.py','.rs','.patch'}:
        shutil.copy2(path,out.with_name(out.name+'.txt'))
    else:
        shutil.copy2(path,out)
inventory={str(p.relative_to(dest)):{'bytes':p.stat().st_size,'sha256':sha(p)}
           for p in sorted(dest.rglob('*')) if p.is_file() and p.name!='inventory.json'}
(dest/'inventory.json').write_text(json.dumps(inventory,indent=2)+'\n')
for name,row in inventory.items():
    assert sha(dest/name)==row['sha256']
summary={'files':len(inventory),'bytes':sum(r['bytes'] for r in inventory.values()),
         'tests':re.search(r'Summary \[.*?(\d+) tests run: \1 passed',check).group(1)}
(scratch/'collection-summary.json').write_text(json.dumps(summary,indent=2)+'\n')
print(json.dumps(summary))
