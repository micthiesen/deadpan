from pathlib import Path
import concurrent.futures,hashlib,json,subprocess,time
r=Path(__file__).parent;raw=r/'quality-reference-full.i420'
d=json.loads((r/'packaged-final/youtube-av1-verify.stdout').read_text())
command=[str(r/'extract_reference_resume'),str(r/'packaged-final/work/youtube-av1.deadpan'),d['revision_id'],str(raw)]
start=time.monotonic()
with (r/'extract-reference-resume.log').open('w') as log:
    result=subprocess.run(command,stdout=log,stderr=subprocess.STDOUT,timeout=1200)
report=dict(command=command,exit=result.returncode,seconds=time.monotonic()-start)
if result.returncode==0:
    assert raw.stat().st_size==3506*3110400
    h=hashlib.sha256()
    with raw.open('rb') as f:
        while chunk:=f.read(1024*1024):h.update(chunk)
    report.update(raw_sha256=h.hexdigest(),bytes=raw.stat().st_size)
report['binary_sha256']=hashlib.sha256((r/'extract_reference_resume').read_bytes()).hexdigest()
(r/'extract-reference-resume.json').write_text(json.dumps(report,indent=2)+'\n')
print('REFERENCES',json.dumps(report),flush=True)
if result.returncode:raise SystemExit(result.returncode)
rows=[]
def run(name,variant,mode=0):
    command=[str(r/('quality-probe-full-'+variant)),str(raw),str(r/(name+'.mp4')),'0','0',str(mode)]
    start=time.monotonic();p=subprocess.run(command,capture_output=True,text=True,timeout=360)
    (r/(name+'.stdout')).write_text(p.stdout);(r/(name+'.stderr')).write_text(p.stderr)
    row=dict(name=name,command=command,exit=p.returncode,seconds=time.monotonic()-start)
    if p.returncode==0:row['result']=json.loads(p.stdout)
    print(json.dumps(row),flush=True);return row
for variant in ['baseline','qmax24']:
    rows.append(run('quality-full-single-'+variant,variant))
    with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
        jobs=[pool.submit(run,'quality-full-parallel-'+variant+'-'+str(index),variant) for index in range(2)]
        rows.extend(job.result() for job in jobs)
    (r/'quality-full-matrix.json').write_text(json.dumps(rows,indent=2)+'\n')
rows.append(run('quality-full-software-qmax24','qmax24',1))
(r/'quality-full-matrix.json').write_text(json.dumps(rows,indent=2)+'\n')
