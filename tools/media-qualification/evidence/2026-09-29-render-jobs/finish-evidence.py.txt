files = []
for path in sorted(DEST.iterdir()):
    if path.is_file() and path.name != 'file-manifest.json':
        with path.open('rb') as file:
            digest = hashlib.file_digest(file, 'sha256').hexdigest()
        files.append({'path': path.name, 'bytes': path.stat().st_size, 'sha256': digest})
(DEST / 'file-manifest.json').write_text(json.dumps(files, indent=2) + '\n')
for row in files:
    path = DEST / row['path']
    assert path.stat().st_size == row['bytes']
    with path.open('rb') as file:
        assert hashlib.file_digest(file, 'sha256').hexdigest() == row['sha256']
print(json.dumps({'files_rehashed': len(files), 'tests': summary['workspace']['passed'], 'archive': audit}))
