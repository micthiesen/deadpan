import hashlib, json, shutil, sqlite3
from pathlib import Path
root = Path('/tmp/deadpan-publication-recovery-ams9kpds')
prior = Path('/tmp/deadpan-render-jobs-95xpekq5')
new = root / 'native-inputs'
new.mkdir()
records = []
for source_name, case in [('source', 'structural'), ('generated', 'generated')]:
    source = prior / 'native-inputs' / (source_name + '.deadpan')
    backup = prior / 'verified-databases' / (case + '.sqlite')
    target = new / (source_name + '.deadpan')
    def ignore(directory, names):
        if Path(directory) == source:
            return [n for n in names if n.startswith('project.sqlite') or n == '.writer.lock']
        return []
    shutil.copytree(source, target, ignore=ignore)
    with sqlite3.connect(f'file:{backup}?mode=ro', uri=True) as src:
        assert src.execute('PRAGMA user_version').fetchone()[0] == 40
        with sqlite3.connect(target / 'project.sqlite') as dst:
            src.backup(dst)
    records.append({'source': str(source), 'source_backup': str(backup), 'source_backup_sha256': hashlib.sha256(backup.read_bytes()).hexdigest(), 'target': str(target), 'method': 'immutable package objects copied, coherent source backup copied through SQLite backup API', 'schema':40})
(root / 'native-inputs.json').write_text(json.dumps(records, indent=2) + '\n')
print(json.dumps(records, indent=2))
