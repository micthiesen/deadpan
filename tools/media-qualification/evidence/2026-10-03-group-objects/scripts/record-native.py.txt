import hashlib
import json
from pathlib import Path

root = Path("/tmp/deadpan-group-objects-20261003")
document = json.loads((root / "native-project.txt").read_text())
bank = json.loads((root / "native-register-a.txt").read_text())
groups = [(key, node) for key, node in document["nodes"].items() if node["label"] == "Native object"]
assert len(groups) == 1
group, node = groups[0]
assert document["nodes"][document["root"]]["kind"]["children"] == [group]
children = node["kind"]["children"]
assert len(children) == 1 and len(document["nodes"]) == 3
copy = bank["registers"][0]
assert copy["register"] == "a" and copy["type"] == "edited" and copy["parent"] == group
assert copy["selection"] == {"type": "children", "first": children[0], "last": children[0]}
session = json.loads((root / "checks/native-session.json").read_text())
assert session["exit_code"] == 0 and session["source_unchanged"]

    "source_manifest_sha256": session["source_manifest_sha256"],
    "project": str(root / "groups-visual-2/projects/groups/Documents/Deadpan/cfr-bframes.deadpan"),
    "input": "CUA native keyboard events with accessibility observations after each edit",
    "observations": [
        "Created Native object from the complete 120-frame Original through native command entry.",
        "Copied group contents with named register a and yig; entering the group preserved the saved copy.",

        "Place r showed exact source endpoints and an unsaved replacement; Escape restored the finished Object, group scope and cursor 119.",

        "dig from inside retained an empty Native object and zero-frame scope; vig retained explicit empty contents; p inserted the copied 120 frames into that group.",
        "One Undo restored empty contents; another restored the cut source in its original group.",
        "Native vag, :splice, r and Return committed the prepared replacement in the outer scope; one Undo restored the group.",

        "Closed-project validation passed; exact Children capture in register a matches the restored named group and child identities."
