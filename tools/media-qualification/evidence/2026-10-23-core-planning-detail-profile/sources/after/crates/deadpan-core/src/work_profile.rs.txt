//! Temporary numeric diagnostic buffer for an isolated source copy only.
//! The app owns activation policy and output.

use std::cell::RefCell;
use std::time::Instant;

#[derive(Clone, Copy)]
pub enum Phase {
    AppServiceTotal,
    AppSemanticCall,
    AppPublication,
    AppRefreshSucceeded,
    SemanticTotal,
    SemanticEntrySetup,
    SemanticExecute,
    SemanticLeafApplyCall,
    SemanticAfterLeafSize,
    SemanticOuterFinish,
    ReplayTotal,
    ReplayHostVisit,
    ReplayWireSizeChecks,
    ReplayLeafApplyCall,
    ReplayAggregateCall,
    LeafApplyTotal,
    LeafPreflight,
    LeafReduce,
    LeafPostprocess,
    LeafNetCall,
    NetTotal,
    NetBeforeAndPatch,
    NetValidation,
    NetFinish,
}

pub const PHASE_COUNT: usize = 24;

#[derive(Default)]
pub struct Measurement {
    pub elapsed_ns: u128,
    pub count: u64,
}

thread_local! {
    static ACTIVE: RefCell<Option<[Measurement; PHASE_COUNT]>> = const { RefCell::new(None) };
}

pub fn begin() -> bool {
    ACTIVE.with(|active| {
        let mut active = active.borrow_mut();
        if active.is_some() {
            return false;
        }
        *active = Some(std::array::from_fn(|_| Measurement::default()));
        true
    })
}

pub struct Span {
    phase: Phase,
    started: Option<Instant>,
}

impl Span {
    pub fn start(phase: Phase) -> Self {
        let started = ACTIVE.with(|active| active.borrow().is_some().then(Instant::now));
        Self { phase, started }
    }
}

impl Drop for Span {
    fn drop(&mut self) {
        let Some(started) = self.started else {
            return;
        };
        record(self.phase, started.elapsed().as_nanos());
    }
}

fn record(phase: Phase, elapsed_ns: u128) {
    ACTIVE.with(|active| {
        if let Some(measurements) = active.borrow_mut().as_mut() {
            let measurement = &mut measurements[phase as usize];
            measurement.elapsed_ns = measurement.elapsed_ns.saturating_add(elapsed_ns);
            measurement.count = measurement.count.saturating_add(1);
        }
    });
}

pub fn mark_refresh_success() {
    record(Phase::AppRefreshSucceeded, 0);
}

pub fn take() -> Option<[Measurement; PHASE_COUNT]> {
    ACTIVE.with(|active| active.borrow_mut().take())
}
