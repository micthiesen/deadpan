fn scoped_macro_target(
    workspace: &Workspace,
    id: &Id,
    scope: &SequenceScope,
    context: &deadpan_core::SemanticContext,
) -> Result<Option<crate::project::scoped::Target>> {
    if let Some(authority) = &context.selected_hold_duration {
        if context.mixed_hold_duration.is_some()
            || context.shared_hold_duration.is_some()
            || context.hold_duration.is_some()
            || context.scoped.is_some()
            || context.original.is_some()
            || context.visual_selection.is_some()
            || context.selected_child.as_ref() != Some(&authority.root)
        {
            return Err(
                "Several-play pause length requires its captured Hold selection without another selection.".into(),
            );
        }
        let target = crate::project::scoped::Target {
            session: id.session,
            navigation: 0,
            project: id.project.clone(),
            revision: id.revision.clone(),
            scope: scope.clone(),
            root: authority.root.clone(),
            target: authority.target.clone(),
            presentation: authority.presentation.clone(),
            cursor: context.cursor,
            also: authority.also.clone(),
        };
        target.validate_request(workspace, id.session, &id.revision, scope, context.cursor)?;
        if target.semantic_selected_hold_duration(workspace)? != **authority {
            return Err(
                "Several-play pause length differs from its captured native authority.".into(),
            );
        }
        return Ok(Some(target));
    }
    if let Some(authority) = &context.mixed_hold_duration {
        if context.shared_hold_duration.is_some()
            || context.hold_duration.is_some()
            || context.scoped.is_some()
            || context.original.is_some()
            || context.visual_selection.is_some()
            || context.selected_child.as_ref() != Some(&authority.root)
        {
            return Err(
                "Mixed pause length requires its captured Hold without another selection.".into(),
            );
        }
        let target = crate::project::scoped::Target {
            session: id.session,
            navigation: 0,
            project: id.project.clone(),
            revision: id.revision.clone(),
            scope: scope.clone(),
            root: authority.root.clone(),
            target: authority.target.clone(),
            presentation: authority.presentation.clone(),
            cursor: context.cursor,
            also: Vec::new(),
        };
        target.validate_request(workspace, id.session, &id.revision, scope, context.cursor)?;
        if target.semantic_mixed_hold_duration(workspace)? != **authority {
            return Err("Mixed pause length differs from its captured native authority.".into());
        }
        return Ok(Some(target));
    }
    if let Some(authority) = &context.shared_hold_duration {
        if context.hold_duration.is_some()
            || context.scoped.is_some()
            || context.original.is_some()
            || context.visual_selection.is_some()
            || context.selected_child.as_ref() != Some(&authority.root)
        {
            return Err("Shared pause length requires its captured All plays Hold without another selection.".into());
        }
        let target = crate::project::scoped::Target {
            session: id.session,
            navigation: 0,
            project: id.project.clone(),
            revision: id.revision.clone(),
            scope: scope.clone(),
            root: authority.root.clone(),
            target: authority.target.clone(),
            presentation: authority.presentation.clone(),
            cursor: context.cursor,
            also: Vec::new(),
        };
        target.validate_request(workspace, id.session, &id.revision, scope, context.cursor)?;
        if target.semantic_shared_hold_duration(workspace)? != *authority {
            return Err("Shared pause length differs from its captured native authority.".into());
        }
        return Ok(Some(target));
    }
    if let Some(authority) = &context.hold_duration {
        if context.scoped.is_some()
            || context.original.is_some()
            || context.visual_selection.is_some()
            || context.selected_child.as_ref() != Some(&authority.root)
        {
            return Err(
                "Ordinary pause length requires its captured Edit Hold without another selection."
                    .into(),
            );
        }
        let target = crate::project::scoped::Target {
            session: id.session,
            navigation: 0,
            project: id.project.clone(),
            revision: id.revision.clone(),
            scope: scope.clone(),
            root: authority.root.clone(),
            target: deadpan_core::ScopedNodeTarget {
                node: authority.target.clone(),
                repeats: Vec::new(),
            },
            presentation: Some(authority.presentation.clone()),
            cursor: context.cursor,
            also: Vec::new(),
        };
        target.validate_request(workspace, id.session, &id.revision, scope, context.cursor)?;
        if target.semantic_hold_duration(workspace)? != *authority {
            return Err("Ordinary pause length differs from its captured native authority.".into());
        }
        return Ok(Some(target));
    }
    let Some(scoped) = &context.scoped else {
        return Ok(None);
    };
    if context.original.is_some() || context.visual_selection.is_some() {
        return Err("One-play macros require Your edit with no Visual selection.".into());
    }
    let target = crate::project::scoped::Target {
        session: id.session,
        navigation: 0,
        project: id.project.clone(),
        revision: id.revision.clone(),
        scope: scope.clone(),
        root: scoped.root.clone(),
        target: scoped.target.clone(),
        presentation: Some(scoped.presentation.clone()),
        cursor: context.cursor,
        also: Vec::new(),
    };
    target.validate_request(workspace, id.session, &id.revision, scope, context.cursor)?;
    if target.semantic_framing(workspace)? != *scoped {
        return Err("The macro authority differs from its captured native scope.".into());
    }
    Ok(Some(target))
}

fn scoped_macro_receipt(
    document: &deadpan_core::ProjectDocument,
    before: Option<crate::project::scoped::Target>,
    context: &deadpan_core::SemanticContext,
    plan: &SemanticPlan,
) -> Result<Option<crate::project::scoped::Commit>> {
    if let Some(authority) = &context.selected_hold_duration {
        let before = before.ok_or("Several-play pause length lost its native capture.")?;
        let request = plan
            .request
            .as_ref()
            .ok_or("Several-play pause length has no executed edit.")?;
        let proof = deadpan_core::derive_scoped_isolation(document, request, &plan.document)
            .map_err(display)?;
        let mapped = deadpan_core::SemanticSelectedHoldDurationContext {
            root: authority.root.clone(),
            target: proof.map_forward(&authority.target).map_err(display)?,
            presentation: authority
                .presentation
                .as_ref()
                .map(|instance| proof.map_instance_forward(instance))
                .transpose()
                .map_err(display)?,
            also: authority
                .also
                .iter()
                .map(|target| proof.map_forward(target).map_err(display))
                .collect::<Result<Vec<_>>>()?,
        };
        let visible = plan
            .document
            .selected_hold_duration_visibility(&context.parent, &mapped)
            .map_err(display)?;
        let mut expected = context.clone();
        expected.selected_hold_duration = Some(Box::new(mapped.clone()));
        if let Some(visible) = visible {
            expected.cursor = visible.frames.start();
            let instance = mapped
                .presentation
                .as_ref()
                .ok_or("Selected pause lost its primary presentation.")?;
            let pictures = RenderPlan::compile(&plan.document).map_err(display)?;
            for frame in [
                visible.frames.start(),
                deadpan_core::ProjectFrame(visible.frames.end().0 - 1),
            ] {
                if !pictures
                    .picture(frame)
                    .map_err(display)?
                    .framing
                    .iter()
                    .any(|layer| &layer.instance == instance)
                {
                    return Err(
                        "Resized selected pause no longer presents its captured primary Hold."
                            .into(),
                    );
                }
            }
        }
        if expected != plan.context {
            return Err(
                "Several-play pause length lost its executed isolation or exact continuation."
                    .into(),
            );
        }
        return Ok(Some(crate::project::scoped::Commit {
            before,
            revision: plan.document.revision_id().clone(),
            target: mapped.target,
            presentation: mapped.presentation,
            also: Some(mapped.also),
        }));
    }
    if let Some(authority) = &context.mixed_hold_duration {
        let before = before.ok_or("Mixed pause length lost its native capture.")?;
        let request = plan
            .request
            .as_ref()
            .ok_or("Mixed pause length has no executed edit.")?;
        let proof = deadpan_core::derive_scoped_isolation(document, request, &plan.document)
            .map_err(display)?;
        let mapped = deadpan_core::SemanticMixedHoldDurationContext {
            root: authority.root.clone(),
            target: proof.map_forward(&authority.target).map_err(display)?,
            presentation: authority
                .presentation
                .as_ref()
                .map(|instance| proof.map_instance_forward(instance))
                .transpose()
                .map_err(display)?,
        };
        let visible = plan
            .document
            .mixed_hold_duration_visibility(&context.parent, &mapped)
            .map_err(display)?;
        let mut expected = context.clone();
        expected.mixed_hold_duration = Some(Box::new(mapped.clone()));
        if let Some(visible) = visible {
            expected.cursor = visible.frames.start();
            let instance = mapped
                .presentation
                .as_ref()
                .ok_or("Mixed pause lost its presentation.")?;
            let pictures = RenderPlan::compile(&plan.document).map_err(display)?;
            for frame in [
                visible.frames.start(),
                deadpan_core::ProjectFrame(visible.frames.end().0 - 1),
            ] {
                if !pictures
                    .picture(frame)
                    .map_err(display)?
                    .framing
                    .iter()
                    .any(|layer| &layer.instance == instance)
                {
                    return Err("Resized mixed pause no longer presents its captured Hold.".into());
                }
            }
        }
        if expected != plan.context {
            return Err(
                "Mixed pause length lost its executed isolation or exact continuation.".into(),
            );
        }
        return Ok(Some(crate::project::scoped::Commit {
            before,
            revision: plan.document.revision_id().clone(),
            target: mapped.target,
            presentation: mapped.presentation,
            also: None,
        }));
    }
    if let Some(authority) = &context.shared_hold_duration {
        let before = before.ok_or("Shared pause length lost its native capture.")?;
        let visible = plan
            .document
            .shared_hold_duration_visibility(&context.parent, authority)
            .map_err(display)?;
        let mut expected = context.clone();
        if let Some(visible) = visible {
            expected.cursor = visible.frames.start();
            let instance = authority
                .presentation
                .as_ref()
                .ok_or("Shared pause lost its presentation.")?;
            let picture_plan = RenderPlan::compile(&plan.document).map_err(display)?;
            for frame in [
                visible.frames.start(),
                deadpan_core::ProjectFrame(visible.frames.end().0 - 1),
            ] {
                if !picture_plan
                    .picture(frame)
                    .map_err(display)?
                    .framing
                    .iter()
                    .any(|layer| &layer.instance == instance)
                {
                    return Err("Resized shared pause no longer presents its captured Hold.".into());
                }
            }
        }
        if expected != plan.context {
            return Err(
                "Shared pause length lost its captured choices or exact continuation.".into(),
            );
        }
        return Ok(Some(crate::project::scoped::Commit {
            target: before.target.clone(),
            presentation: authority.presentation.clone(),
            before,
            revision: plan.document.revision_id().clone(),
            also: None,
        }));
    }
    if let Some(authority) = &context.hold_duration {
        let before = before.ok_or("Ordinary pause length lost its native capture.")?;
        let visible = plan
            .document
            .ordinary_hold_duration_visibility(&context.parent, authority)
            .map_err(display)?;
        let mut expected = context.clone();
        expected.cursor = visible.frames.start();
        if expected != plan.context {
            return Err(
                "Ordinary pause length lost its captured authority or visible continuation.".into(),
            );
        }
        let picture_plan = RenderPlan::compile(&plan.document).map_err(display)?;
        for frame in [
            visible.frames.start(),
            deadpan_core::ProjectFrame(visible.frames.end().0 - 1),
        ] {
            let picture = picture_plan.picture(frame).map_err(display)?;
            if !picture
                .framing
                .iter()
                .any(|layer| layer.instance == authority.presentation)
            {
                return Err("Resized pause no longer presents its captured Hold.".into());
            }
        }
        return Ok(Some(crate::project::scoped::Commit {
            target: before.target.clone(),
            presentation: Some(authority.presentation.clone()),
            before,
            revision: plan.document.revision_id().clone(),
            also: None,
        }));
    }
    match (before, &plan.context.scoped) {
        (None, None) => Ok(None),
        (Some(before), Some(after)) => {
            let mut expected = context.clone();
            expected.scoped = Some(after.clone());
            // A fixed window continues at its first visible sampled picture;
            // ordinary occurrence edits retain their intrinsic Hold start.
            // Keep every other ordinary context field strict.
            if plan
                .trace
                .iter()
                .any(|row| matches!(row.instruction, SemanticInstruction::SetHoldDuration { .. }))
            {
                let fixed = crate::project::scoped::fixed_duration_visibility(
                    &plan.document,
                    &context.parent,
                    after,
                )?;
                let valid_start = if let Some(visible) = fixed {
                    let picture_plan = RenderPlan::compile(&plan.document).map_err(display)?;
                    crate::project::scoped::verify_duration_pictures(
                        &picture_plan,
                        &after.presentation,
                        visible,
                    )?;
                    visible.frames.start() == plan.context.cursor
                } else {
                    let start = deadpan_core::AnchorIndex::new(&plan.document)
                        .map_err(display)?
                        .resolve_target(&deadpan_core::AnchorTarget {
                            boundary: deadpan_core::BoundaryAnchor {
                                coordinate: deadpan_core::Anchor::Occurrence {
                                    instance: after.presentation.clone(),
                                    position: deadpan_core::ExactRatio::ZERO,
                                },
                                bias: deadpan_core::InsertionBias::Right,
                            },
                            occurrence: None,
                        })
                        .map_err(display)?;
                    start.exact_frame == deadpan_core::ExactRatio::integer(plan.context.cursor.0)
                };
                if !matches!(
                    plan.document
                        .nodes()
                        .get(&after.target.node)
                        .map(|node| &node.kind),
                    Some(deadpan_core::NodeKind::Hold { .. })
                ) || !valid_start
                {
                    return Err(
                        "The scoped pause length lost its exact Hold-start continuation.".into(),
                    );
                }
                expected.cursor = plan.context.cursor;
            }
            if expected != plan.context
                || before.root != after.root
                || !after
                    .target
                    .matches_instance(&plan.document, &after.presentation)
                    .map_err(display)?
            {
                return Err(
                    "The scoped macro changed its ordinary context or lost its concrete play."
                        .into(),
                );
            }
            Ok(Some(crate::project::scoped::Commit {
                before,
                revision: plan.document.revision_id().clone(),
                target: after.target.clone(),
                presentation: Some(after.presentation.clone()),
                also: None,
            }))
        }
        _ => Err("The scoped macro did not preserve its invocation authority.".into()),
    }
}

impl Service {
    /// Remote macros use explicit document coordinates. Build every runtime
    /// copy before committing, without creating a native cursor continuation.
    pub(super) fn prepare_remote_macro_registers(
        &mut self,
        prepared: &deadpan_cli::macros::Prepared,
    ) -> Result<Option<Arc<Bank>>> {
        use deadpan_cli::macros::Operation as RemoteOperation;

        if prepared.request.dry_run
            || prepared
                .plan
                .as_ref()
                .is_some_and(|plan| plan.request.is_none())
        {
            return Ok(None);
        }
        self.refresh_registers()?;
        let id = Id {
            session: self.session,
            project: prepared.request.project_id.clone(),
            revision: prepared.request.expected_revision.clone(),
            bank_version: prepared.bank.version,
            // Used only to prepare persisted runtime copies. No native request
            // or receipt is published for this remote operation.
            request: 1,
            transcript_guard: None,
        };
        let runtime = self.macro_runtime_bank(&id)?;
        if runtime.entries.len() != prepared.bank.entries.len()
            || prepared.bank.entries.iter().any(|(name, value)| {
                runtime
                    .entries
                    .get(&name.as_char())
                    .is_none_or(|runtime| !matches_runtime(value, runtime, &id))
            })
        {
            return Err("Remote macro input differs from its runtime register content".into());
        }
        let bank = match &prepared.request.operation {
            RemoteOperation::Save { register, program } => {
                let mut bank = runtime.clone();
                bank.version = prepared.final_bank.version;
                bank.entries
                    .insert(register.as_char(), Value::Macro(program.clone()));
                bank
            }
            RemoteOperation::YankOriginal { register, .. } => {
                // The same runtime value a native `y` in Original saves.
                let Some(deadpan_core::RegisterValue::Original {
                    asset,
                    qualification,
                    ordinals,
                    ..
                }) = prepared.final_bank.entries.get(register).map(AsRef::as_ref)
                else {
                    return Err("Remote Original copy has no prepared value".into());
                };
                let value = Value::Original {
                    asset: asset.clone(),
                    qualification: qualification.clone(),
                    ordinals: ordinals.clone(),
                };
                let mut bank = runtime.clone();
                bank.version = prepared.final_bank.version;
                bank.entries.insert('"', value.clone());
                bank.entries.insert(register.as_char(), value);
                bank
            }
            RemoteOperation::Run { .. } | RemoteOperation::Apply { .. } => {
                let plan = prepared.plan.as_ref().ok_or("Remote macro has no plan")?;
                prepare_runtime_bank(&id, runtime, &prepared.bank, &prepared.final_bank, plan)?
            }
        };
        Ok(Some(Arc::new(bank)))
    }

    pub(super) fn macro_command(&mut self, operation: Operation) {
        let result = match &self.saved_macro {
            Some((previous, receipt)) if previous == &operation => Ok(receipt.clone()),
            Some((previous, _)) if previous.id() == operation.id() => {
                Err("The macro identity was already used with different parameters".into())
            }
            _ => self.perform_macro(&operation),
        };
        self.macros = Some(Update {
            id: operation.id().clone(),
            result,
        });
    }

    fn perform_macro(&mut self, operation: &Operation) -> Result<Receipt> {
        self.observe_semantic();
        let id = operation.id();
        let (document, stored) = self.check_macro_context(id)?;
        match operation {
            Operation::Save {
                register, program, ..
            } => {
                let name = macro_name(*register)?;
                program.validate().map_err(display)?;
                let mut bank = self.macro_runtime_bank(id)?.clone();
                bank.version = next_bank_version(stored.version)?;
                bank.entries
                    .insert(*register, Value::Macro(program.clone()));
                let saved = self
                    .writer()?
                    .save_macro(
                        &id.project,
                        &id.revision,
                        id.bank_version,
                        name,
                        program.clone(),
                    )
                    .map_err(display)?;
                // The store owns version advancement. All runtime preparation
                // above is fallible; after success publication cannot fail.
                bank.version = saved.version;
                self.registers = Some(Arc::new(bank));
                let receipt = Receipt {
                    transcript_guard: None,
                    id: id.clone(),
                    bank_version: saved.version,
                    outcome: Outcome::Saved {
                        register: *register,
                        instructions: program.instructions().len(),
                    },
                };
                self.saved_macro = Some((operation.clone(), receipt.clone()));
                self.message = Some(format!("Macro {register} saved"));
                self.error = None;
                Ok(receipt)
            }
            Operation::Run { scope, context, .. } | Operation::Apply { scope, context, .. } => {
                if let Operation::Apply {
                    instruction,
                    repeat_version: Some(version),
                    ..
                } = operation
                {
                    self.check_semantic_repeat(*version, context, instruction)?;
                }
                let (instruction, label) = match operation {
                    Operation::Run {
                        register, count, ..
                    } => (
                        SemanticInstruction::Call {
                            register: macro_name(*register)?,
                            count: NonZeroU32::new(*count).ok_or("Macro count must be positive")?,
                        },
                        format!("Macro {register}"),
                    ),
                    Operation::Apply { instruction, .. } => (
                        instruction.as_ref().clone(),
                        match instruction.as_ref() {
                            SemanticInstruction::YankBeat { .. } => "Copy beat",
                            SemanticInstruction::Yank {
                                selector: SemanticSelector::SelectedBeat,
                                ..
                            } => "Copy beat",
                            SemanticInstruction::YankSelection { .. } => "Copy selection",
                            SemanticInstruction::CutFrames { .. } => "Cut frames",
                            SemanticInstruction::CutSelection { .. } => "Cut selection",
                            SemanticInstruction::Cut {
                                selector: SemanticSelector::SelectedBeat,
                                ..
                            } => "Cut beat",
                            SemanticInstruction::Yank { .. } => "Copy selection",
                            SemanticInstruction::Cut { .. } => "Cut selection",
                            SemanticInstruction::ReplaceSelection { .. } => "Replace selection",
                            SemanticInstruction::Paste { .. } => "Paste register",
                            SemanticInstruction::InsertOriginal { .. } => "Reuse whole Original",
                            SemanticInstruction::Repeat { .. } => "Wrap Repeat",
                            SemanticInstruction::SetRepeatPlays { .. } => "Set Repeat plays",
                            SemanticInstruction::SetRepeat { .. } => "Change Repeat",
                            SemanticInstruction::SetRoomTone { .. } => "Set room tone",
                            SemanticInstruction::SetHoldAudio { .. } => "Change pause sound",
                            SemanticInstruction::SetCutaway { .. } => "Place cutaway",
                            SemanticInstruction::SetCaption { .. } => "Add caption",
                            SemanticInstruction::Group { .. } => "Group beats",
                            SemanticInstruction::Ungroup => "Ungroup beats",
                            SemanticInstruction::Explode => "Explode Repeat",
                            SemanticInstruction::Duplicate { .. } => "Duplicate",
                            SemanticInstruction::Gag { .. } => "Apply gag",
                            SemanticInstruction::SetGag { .. } => "Change gag parameters",
                            SemanticInstruction::SetAudioEdges { .. } => "Change sound edges",
                            SemanticInstruction::SetAudioLag { .. } => "Offset sound",
                            SemanticInstruction::InsertPause { .. } => "Insert pause",
                            SemanticInstruction::InsertAiPause { .. } => "Insert AI pause",
                            SemanticInstruction::InsertReverse { bounce: false, .. } => {
                                "Insert reverse"
                            }
                            SemanticInstruction::InsertReverse { bounce: true, .. } => {
                                "Insert ping-pong"
                            }
                            SemanticInstruction::Tail { .. } => "Add tail",
                            SemanticInstruction::Lift { .. } => "Lift selection",
                            SemanticInstruction::Bleep { .. } => "Bleep selection",
                            SemanticInstruction::SetFraming { .. }
                            | SemanticInstruction::Zoom { .. } => "Frame beat",
                            SemanticInstruction::Retime { .. } => "Change speed",
                            SemanticInstruction::Pitch { .. } => "Shift pitch",
                            SemanticInstruction::SetHoldDuration { .. } => "Change pause length",
                            SemanticInstruction::SetAudio {
                                change: deadpan_core::AudioChange::Saturation { .. },
                            } => "Change saturation",
                            SemanticInstruction::SetAudio {
                                change: deadpan_core::AudioChange::Balance { .. },
                            } => "Change stereo balance",
                            SemanticInstruction::SetAudio {
                                change: deadpan_core::AudioChange::Equalizer { .. },
                            } => "Change EQ",
                            SemanticInstruction::SetAudio {
                                change: deadpan_core::AudioChange::TailSend { .. },
                            } => "Change tail send",
                            SemanticInstruction::SetAudio {
                                change: deadpan_core::AudioChange::WetSend { .. },
                            } => "Change wet return",
                            SemanticInstruction::SetAudio { .. } => "Change gain",
                            SemanticInstruction::RoleRepeat {
                                role: deadpan_core::MediaRole::Audio,
                                ..
                            } => "Repeat sound",
                            SemanticInstruction::RoleRepeat { .. } => "Repeat picture",
                            SemanticInstruction::DeleteRole {
                                role: deadpan_core::MediaRole::Audio,
                            } => "Delete sound",
                            SemanticInstruction::DeleteRole { .. } => "Delete picture",
                            SemanticInstruction::SplitEdit { kind, .. } => match kind {
                                deadpan_core::SplitEditKind::J => "J-cut",
                                deadpan_core::SplitEditKind::L => "L-cut",
                            },
                            _ => "Recorded action",
                        }
                        .to_owned(),
                    ),
                    Operation::Save { .. } => unreachable!("save handled separately"),
                };
                let picture_plan = RenderPlan::compile(&document).map_err(display)?;
                let owner = scope.resolve_document(&document, &picture_plan)?;
                if owner.owner != &context.parent {
                    return Err("Macro target differs from the captured Sequence scope".into());
                }
                let scoped_before = scoped_macro_target(
                    self.workspace.as_ref().ok_or("Open a project first")?,
                    id,
                    scope,
                    context,
                )?;
                let invocation = SemanticProgram::new(vec![instruction]).map_err(display)?;
                pause_route_profile::set(PausePhase::SemanticPlan);
                let plan = deadpan_cli::macros::plan_program_with_transcript_guard(
                    self.store.as_ref().ok_or("Open a project first")?,
                    &document,
                    &stored,
                    context,
                    &invocation,
                    revision(),
                    id.transcript_guard.as_deref(),
                )
                .map_err(display)?;
                pause_route_profile::set(PausePhase::NativeContinuation);
                if let Some(consumed) = &plan.transcript_guard {
                    let displayed = self
                        .workspace
                        .as_ref()
                        .and_then(|workspace| workspace.transcript.as_ref())
                        .and_then(|transcript| transcript.selection_guard.as_ref());
                    if id.transcript_guard.as_deref() != Some(consumed)
                        || displayed != Some(consumed)
                    {
                        return Err("The displayed transcript changed or was unavailable when this command was entered. Enter the command again.".into());
                    }
                }
                let final_picture_plan = RenderPlan::compile(&plan.document).map_err(display)?;
                let final_scope = SequenceScope::from_historical_parent(
                    &plan.document,
                    &final_picture_plan,
                    &plan.context.parent,
                )?;
                let scoped_receipt =
                    scoped_macro_receipt(&document, scoped_before, context, &plan)?;
                let mut receipt = Receipt {
                    transcript_guard: plan.transcript_guard.clone(),
                    id: id.clone(),
                    bank_version: stored.version,
                    outcome: match operation {
                        Operation::Run {
                            register, count, ..
                        } => Outcome::Executed {
                            register: *register,
                            count: *count,
                            scope: final_scope.clone(),
                            cursor: plan.context.cursor,
                            selected: plan.selected_child.clone(),
                            visual_selection: plan.context.visual_selection.clone(),
                            original: plan.context.original.clone(),
                            committed: None,
                            refresh_error: None,
                        },
                        Operation::Apply { .. } => Outcome::Applied {
                            scope: final_scope.clone(),
                            cursor: plan.context.cursor,
                            selected: plan.selected_child.clone(),
                            visual_selection: plan.context.visual_selection.clone(),
                            original: plan.context.original.clone(),
                            committed: None,
                            refresh_error: None,
                        },
                        Operation::Save { .. } => unreachable!("save handled separately"),
                    },
                };
                let Some(_) = &plan.request else {
                    plan.validate_selection_at(
                        self.store.as_ref().ok_or("Open a project first")?,
                        &id.project,
                        &id.revision,
                        id.bank_version,
                    )
                    .map_err(display)?;
                    self.saved_macro = Some((operation.clone(), receipt.clone()));
                    self.message = Some(if plan.context.original.is_some() {
                        format!("{label} updated the Original cursor")
                    } else {
                        format!("{label} updated the Edit cursor and selection")
                    });
                    self.error = None;
                    return Ok(receipt);
                };
                self.macro_runtime_bank(id)?;
                let runtime =
                    Arc::clone(self.registers.as_ref().ok_or("Missing runtime registers")?);
                // The store prepares the Compound once, under its publication
                // transaction. Runtime copies remain fallible and private until
                // that same prepared edit and bank have durably committed.
                let (outcome, bank) = plan
                    .commit_compound_prepared(self.writer()?, |prepared| {
                        prepare_runtime_bank(id, &runtime, &stored, prepared, &plan)
                    })
                    .map_err(|error| match error {
                        deadpan_cli::macros::PreparedCommitError::Commit(error) => display(error),
                        deadpan_cli::macros::PreparedCommitError::Preparation(error) => error,
                    })?;
                pause_route_profile::set(PausePhase::NativePostcommit);
                if outcome.committed.is_some() {
                    pause_route_profile::committed();
                }
                if let Some(saved) = &outcome.committed {
                    self.capture_preparation_notices(&saved.generation_preparation_notices);
                    if plan.context.original.is_some() {
                        self.preserve_semantic(&id.revision, &saved.revision_id);
                        self.marks.record_metadata(
                            id.session,
                            &id.project,
                            &id.revision,
                            &saved.revision_id,
                        );
                    }
                }
                if let Some(saved) = &outcome.committed
                    && let Operation::Apply { instruction, .. } = operation
                    && let Some(edit) =
                        crate::project::semantic::LastEdit::from_instruction(instruction)
                {
                    self.semantic.prove(
                        id.session,
                        &id.project,
                        &id.revision,
                        &saved.revision_id,
                        semantic::Change::Replace(Box::new(edit)),
                    );
                }
                // A yank-only program saves its bank without a new timeline
                // revision. Only the store's authored receipt grants a native
                // edit continuation; never infer one from a Compound request.
                let committed = outcome.committed.map(|saved| CommittedEdit {
                    selected_node: scoped_receipt
                        .as_ref()
                        .map(|receipt| receipt.before.root.clone())
                        .or_else(|| plan.selected_child.clone()),
                    preserve_cursor: scoped_receipt.is_some(),
                    scoped: scoped_receipt,
                    revision: saved.revision_id,
                    cursor: Some(plan.context.cursor),
                    scope: final_scope,
                    sound: None,
                    range_selection: None,
                });
                receipt.bank_version = outcome.register_bank.version;
                if let Outcome::Executed {
                    committed: result, ..
                }
                | Outcome::Applied {
                    committed: result, ..
                } = &mut receipt.outcome
                {
                    *result = committed.clone().map(Box::new);
                }
                self.registers = Some(Arc::new(bank));
                if let Some(committed) = &committed {
                    // The Macro receipt retains durable success. Original
                    // marks grant no generic Edit continuation, including in
                    // later publications after a bank save replaces that receipt.
                    self.committed = plan.context.original.is_none().then(|| committed.clone());
                }
                self.saved_macro = Some((operation.clone(), receipt.clone()));
                #[cfg(test)]
                {
                    self.render_preview_refresh_failure = self
                        .shared
                        .render_commit_refresh_failure
                        .swap(false, Ordering::AcqRel);
                }
                pause_route_profile::set(PausePhase::WorkspaceRefresh);
                match self.refresh() {
                    Ok(()) => {
                        pause_route_profile::set(PausePhase::NativeFinish);
                        pause_route_profile::refreshed();
                        self.message = Some(if committed.is_some() {
                            format!("{label} saved as one edit; Undo restores it")
                        } else {
                            format!("{label} saved to registers; timeline history is unchanged")
                        })
                    }
                    Err(error) => {
                        pause_route_profile::set(PausePhase::NativeFinish);
                        let message = format!(
                            "{label} saved, but the preview could not refresh: {error}. Reopen this project before editing or undoing."
                        );
                        if let Outcome::Executed { refresh_error, .. }
                        | Outcome::Applied { refresh_error, .. } = &mut receipt.outcome
                        {
                            *refresh_error = Some(message.clone());
                        }
                        self.message = Some(message);
                    }
                }
                self.error = None;
                self.saved_macro = Some((operation.clone(), receipt.clone()));
                Ok(receipt)
            }
        }
    }

    fn check_macro_context(&mut self, id: &Id) -> Result<(ProjectDocument, RegisterBank)> {
        if id.session == 0 || id.request == 0 {
            return Err("Macro identities must be nonzero".into());
        }
        self.check_context(id.session, &id.revision)?;
        if self
            .workspace
            .as_ref()
            .is_none_or(|workspace| workspace.document.project_id() != &id.project)
        {
            return Err("Macro belongs to another project".into());
        }
        if self.pending_session_change.is_some() {
            return Err("Project session is changing; start the macro again after it opens".into());
        }
        let store = self.store.as_ref().ok_or("Open a project first")?;
        let document = store.snapshot().map_err(display)?;
        if document.project_id() != &id.project {
            return Err("Macro belongs to another project".into());
        }
        if document.revision_id() != &id.revision {
            return Err(
                "The saved project changed. Reopen it before running or saving macros.".into(),
            );
        }
        let stored = store.registers().map_err(display)?;
        if stored.version != id.bank_version {
            return Err("The register bank changed; start the macro request again".into());
        }
        self.refresh_registers()?;
        let runtime = self.macro_runtime_bank(id)?;
        if runtime.entries.len() != stored.entries.len()
            || stored.entries.iter().any(|(name, value)| {
                runtime
                    .entries
                    .get(&name.as_char())
                    .is_none_or(|runtime| !matches_runtime(value, runtime, id))
            })
        {
            return Err("Macro input differs from its runtime register content".into());
        }
        Ok((document, stored))
    }

    fn macro_runtime_bank(&self, id: &Id) -> Result<&Bank> {
        self.registers
            .as_deref()
            .filter(|bank| {
                bank.session == id.session
                    && bank.project == id.project
                    && bank.version == id.bank_version
            })
            .ok_or_else(|| "The runtime register bank differs from the macro request".into())
    }
}

fn macro_name(register: char) -> Result<RegisterName> {
    if !register.is_ascii_lowercase() {
        return Err("A macro requires a named register a-z".into());
    }
    RegisterName::new(register).map_err(display)
}

fn next_bank_version(version: u64) -> Result<u64> {
    version
        .checked_add(1)
        .filter(|version| *version <= i64::MAX as u64)
        .ok_or_else(|| "Register versions are exhausted".into())
}

fn matches_runtime(value: &RegisterValue, runtime: &Value, id: &Id) -> bool {
    match (value, runtime) {
        (
            RegisterValue::Original {
                asset,
                qualification,
                ordinals,
                ..
            },
            Value::Original {
                asset: found_asset,
                qualification: found_qualification,
                ordinals: found_ordinals,
            },
        ) => {
            asset == found_asset
                && qualification == found_qualification
                && ordinals == found_ordinals
        }
        (RegisterValue::Edited { slice }, Value::Edited(copied)) => {
            copied.id().session == id.session
                && copied.id().project == id.project
                && copied.id().source_revision == *slice.revision_id()
                && copied.slice() == slice
        }
        (RegisterValue::EditedSet { set }, Value::EditedSet(found)) => set == found,
        (RegisterValue::Macro { program }, Value::Macro(found)) => program == found,
        _ => false,
    }
}

fn prepare_runtime_bank(
    id: &Id,
    runtime: &Bank,
    stored: &RegisterBank,
    preview: &RegisterBank,
    plan: &SemanticPlan,
) -> Result<Bank> {
    let mut expected = stored.entries.clone();
    expected.extend(plan.register_writes.clone());
    let expected_version = if plan.register_writes.is_empty() {
        stored.version
    } else {
        next_bank_version(stored.version)?
    };
    if preview.entries != expected || preview.version != expected_version {
        return Err("Macro preview register writes differ from the prepared semantic plan".into());
    }
    let mut entries = BTreeMap::new();
    let mut prepared: Vec<(Arc<RegisterValue>, Value)> = Vec::new();
    for (name, value) in &preview.entries {
        let entry = if stored.entries.get(name) == Some(value) {
            runtime
                .entries
                .get(&name.as_char())
                .filter(|entry| matches_runtime(value, entry, id))
                .ok_or("Macro input differs from its runtime register content")?
                .clone()
        } else if let Some((_, prepared)) = prepared.iter().find(|(previous, _)| previous == value)
        {
            prepared.clone()
        } else if let Some((slot, _)) = stored
            .entries
            .iter()
            .find(|(_, previous)| *previous == value)
        {
            let existing = runtime
                .entries
                .get(&slot.as_char())
                .filter(|entry| matches_runtime(value, entry, id))
                .ok_or("Macro input differs from its runtime register content")?;
            existing.clone()
        } else if let RegisterValue::EditedSet { set } = value.as_ref() {
            let trace = plan.trace.iter().find(|trace| {
                trace.capture_set.as_ref().is_some_and(|capture| {
                    &capture.parent == set.scope() && &capture.timing == set.capture_timing()
                        && capture.bounds == trace.before_scope
                }) && &trace.before_revision == set.revision_id()
                    && &trace.before.parent == set.scope()
                    && trace.capture.is_none()
                    && trace.resolved_range.is_none()
                    && trace.resolved_parent.is_none()
                    && trace.resolved_selection.is_none()
                    && trace.removed_range.is_none()
                    && trace.captured_child_label.is_none()
                    && trace.resolved_ranges == set.components().iter().map(|component|component.range()).collect::<Vec<_>>()
                    && (
                        (trace.resolved_nodes.len() == set.components().len()
                            && trace.resolved_nodes.iter().zip(set.components()).all(|(member, component)| {
                                component.parent() == &member.parent && component.range() == member.range
                                    && matches!(component.selection(), SliceCaptureSelection::Child { node } if node == &member.node)
                            }))
                        || (trace.resolved_nodes.is_empty()
                            && set.components().iter().all(|component| component.parent() == set.scope()
                                && matches!(component.selection(), SliceCaptureSelection::Range { range } if *range == component.range())))
                    )
            }).ok_or("Macro edited-union copy has no matching staged capture provenance")?;
            if set.project_id() != &id.project {
                return Err("Macro edited-union copy has invalid project provenance".into());
            }
            SequenceScope::from_staged_capture(
                trace
                    .capture_set
                    .as_ref()
                    .ok_or("Missing edited-union capture scope")?,
            )?;
            Value::EditedSet(set.clone())
        } else {
            let RegisterValue::Edited { slice } = value.as_ref() else {
                return Err("Macro plan introduced an unexpected register content type".into());
            };
            let selection = slice.selection();
            let trace = plan
                .trace
                .iter()
                .find(|trace| {
                    trace.capture.as_ref().is_some_and(|capture| {
                        &capture.timing == slice.capture_timing()
                            && &capture.parent == slice.parent()
                    }) && &trace.before_revision == slice.revision_id()
                        && trace.resolved_parent.as_ref() == Some(slice.parent())
                        && trace.resolved_selection.as_ref() == Some(selection)
                })
                .ok_or("Macro copy has no matching staged capture provenance")?;
            let capture = trace
                .capture
                .as_ref()
                .ok_or("Macro copy has no staged capture provenance")?;
            let scope = SequenceScope::from_staged_capture(capture)?;
            let source_path: Vec<String> = std::iter::once("Your edit".to_owned())
                .chain(capture.scope_labels.iter().cloned())
                .collect();
            let child_label = match selection {
                SliceCaptureSelection::Range { .. } | SliceCaptureSelection::Children { .. } => {
                    None
                }
                SliceCaptureSelection::Child { .. } => Some(
                    trace
                        .captured_child_label
                        .clone()
                        .ok_or("Macro beat copy has no staged child label")?,
                ),
            };
            let request = u64::try_from(prepared.len())
                .map_err(display)?
                .checked_add(1)
                .ok_or("Macro copy identities exhausted")?;
            Value::Edited(Arc::new(Captured {
                id: CopyId {
                    session: id.session,
                    project: id.project.clone(),
                    source_revision: slice.revision_id().clone(),
                    request,
                    persisted_version: Some(preview.version),
                },
                scope,
                slice: slice.clone(),
                bounds: capture.bounds,
                source_path,
                child_label,
            }))
        };
        prepared.push((value.clone(), entry.clone()));
        entries.insert(name.as_char(), entry);
    }
    Ok(Bank {
        session: id.session,
        project: id.project.clone(),
        version: preview.version,
        entries,
    })
}
