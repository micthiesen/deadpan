import array
from fractions import Fraction
import hashlib
import json
import math
import os
from pathlib import Path
import subprocess

root = Path('/Users/michael/Code/deadpan')
work = Path('/tmp/deadpan-encoder-admission-20260930')
os.environ['DEADPAN_CHECK_OUTPUT'] = str(work)
runner = root / 'tools/media-qualification/evidence/2026-09-29-generated-pictures/run-native.py'
reader = work / 'avfoundation_probe'

def run(name, args):
    result = subprocess.run(['python3', str(runner), name, *map(str, args)], cwd=root)
    if result.returncode:
        raise RuntimeError((name, result.returncode))

run('avfoundation-build', ['xcrun', '--sdk', 'macosx', 'clang', '-std=c11', '-O2', '-g', '-fobjc-arc', '-Wall', '-Wextra', '-Werror', '-arch', 'arm64', '-mmacosx-version-min=15.0', 'tools/media-qualification/compatible/avfoundation_probe.m', '-o', reader, '-framework', 'Foundation', '-framework', 'AVFoundation', '-framework', 'CoreMedia', '-framework', 'AudioToolbox'])
cases = [('ntsc-final', 320, 180, 30000, 1001), ('sixty', 640, 360, 60, 1),
         ('hd-ntsc', 1920, 1080, 30000, 1001), ('minimum', 14, 16, 30, 1),
         ('low-rate', 640, 360, 4, 1)]
summary = {'cases': [], 'reader_sha256': hashlib.sha256(reader.read_bytes()).hexdigest()}
for name, width, height, num, den in cases:
    case = {'name': name, 'raster': [width, height], 'rate': [num, den]}
    summary['cases'].append(case)
    try:
        report_path = work / (name + '.json')
        run('native-' + name, ['target/debug/examples/qualify_encoder_admission', 'target/debug/deadpan-cli', width, height, num, den, report_path])
        report = json.loads(report_path.read_text())
        assert report['status'] == 'passed'
        selected = report['decision']['selected']
        native = selected['manifest']['report']
        assert native['info']['maximum_moov_bytes'] < 1024 * 1024
        assert selected['spec']['choice'] == {'mode': 'hardware', 'b_frames': 'none'}
        assert len(report['decision']['rejected']) == (0 if num == 4 else 1)
        movie = work / (name + '.probe.mp4')
        pcm = work / (name + '.avfoundation.f32')
        run('avfoundation-' + name, [reader, movie, pcm])
        observed = json.loads((work / ('avfoundation-' + name + '.log')).read_text())
        assert observed['status'] == 'completed'
        assert observed['input_sha256'] == observed['final_input_sha256'] == report['retained_probe']['sha256']
        audio = observed['pcm']
        samples = selected['content']['audio_samples']
        assert audio['total_samples'] == samples
        assert audio['pcm_bytes'] == samples * 8
        assert pcm.stat().st_size == samples * 8
        cursor = 0
        for buffer in audio['buffers']:
            pts = buffer['pts']
            assert Fraction(pts['value'], pts['timescale']) * 48000 == cursor
            assert buffer['sample_offset'] == cursor
            cursor += buffer['num_samples']
        assert cursor == samples
        decoded = array.array('f')
        decoded.frombytes(pcm.read_bytes())
        assert all(math.isfinite(sample) for sample in decoded)
        events = []
        for marker in selected['content']['markers']:
            for channel, expected in enumerate(marker['expected_samples']):
                sign = 1 if channel == 0 else -1
                candidates = range(max(0, expected - 64), min(samples, expected + 65))
                actual = max(candidates, key=lambda index: sign * decoded[index * 2 + channel])
                peak = decoded[actual * 2 + channel]
                assert actual == expected and sign * peak >= .15, (name, channel, expected, actual, peak)
                events.append({'channel': channel, 'expected': expected, 'observed': actual, 'peak': peak})
        case.update(status='passed', frames=native['video_frames'], samples=samples,
                    gops=selected['verification']['gops'], rejected=len(report['decision']['rejected']),
                    maximum_plane_error=selected['content']['maximum_plane_error'],
                    avfoundation_events=events)
    except Exception as error:
        case.update(status='failed', error=str(error))
    (work / 'native-cases-summary.json').write_text(json.dumps(summary, indent=2) + '\n')
run('native-smoke', ['cargo', 'run', '-p', 'deadpan-app', '--locked', '--', '--smoke-test'])
assert all(case['status'] == 'passed' for case in summary['cases']), summary
