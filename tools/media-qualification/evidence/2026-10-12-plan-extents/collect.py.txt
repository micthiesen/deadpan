import gzip, hashlib, json, pathlib, re, subprocess

root = pathlib.Path('/Users/michael/Code/deadpan')
scratch = pathlib.Path(__file__).resolve().parent
dest = root / 'tools/media-qualification/evidence/2026-10-12-plan-extents'
dest.mkdir(parents=True, exist_ok=True)

def sha(path):
    with path.open('rb') as source:
        return hashlib.file_digest(source, 'sha256').hexdigest()

checks, binaries = {}, {}
for receipt in sorted(scratch.glob('*.json')):
    name = receipt.stem
    checks[name] = json.loads(receipt.read_text())
    log = scratch / (name + '.log')
    data = log.read_bytes()
    assert hashlib.sha256(data).hexdigest() == checks[name]['log_sha256']
    (dest / (name + '.log.gz')).write_bytes(gzip.compress(data, mtime=0))
    rows = [tuple(map(int, row)) for row in re.findall(r'test result: (?:ok|FAILED)\. (\d+) passed; (\d+) failed; (\d+) ignored;', data.decode())]
    if rows:
        checks[name]['tests'] = dict(zip(['passed', 'failed', 'ignored'], map(sum, zip(*rows))))
    if name in ['regressions', 'app-camera-bin', 'app-presentation']:
        binaries[name] = {binary: sha(root / binary) for binary in re.findall(
            r'Running .*?\((target/[^ )]+/deadpan_(?:plan|audio|playback|app)-[^ )]+)\)', data.decode())}

sources = json.loads((scratch / 'verified-sources.txt').read_text())
for path, identity in sources.items():
    assert sha(root / path) == identity, path
metadata = {
    'base_commit': subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=root, text=True).strip(),
    'sources_sha256': sources,
    'cargo_lock_sha256': sha(root / 'Cargo.lock'),
    'checks': checks, 'test_binaries_by_check_sha256': binaries,
    'binary_hash_scope': 'Final regression library tests and both native test filters. Earlier focused binaries are not independently identified.',
    'rustc': subprocess.check_output(['rustc', '--version'], cwd=root, text=True).strip(),
    'macos': subprocess.check_output(['sw_vers'], text=True).strip(),
    'hardware': subprocess.check_output(['sysctl', '-n', 'machdep.cpu.brand_string', 'hw.ncpu', 'hw.memsize'], text=True).strip(),
    'review': [
        'Primary implemented and self-reviewed alone.',
        'Reviewed every modified allocation consumer, remaining intrinsic-duration read and the Camera preview path.',
        'Ordinary and retained-context constructors embed the validated integer graph.',
        'Preserve preparation refuses derived allocation until an owned processing route can preserve its intrinsic operand.',
        'Source/Hold mappings and intrinsic counts are not rescaled; independent voices retain explicit natural-rate mappings.',
    ],
    'setup_failures': [
        'The first direct cargo fmt failed because the new test module was in prefixes/tests/extents.rs; it was moved to prefixes/extents.rs.',
        'Two focused builds caught fixture errors: missing DefinitionIndex arguments and a qualification identity requiring a 64-digit hexadecimal String.',
        'app-camera selected the empty library and ran zero tests; app-camera-bin targets the actual binary and runs ten tests.',
    ],
    'limits': 'Compiler-boundary exact allocation and Camera evaluation only. No persisted effective clocks, authored fractional project, new decoded-PCM path, live GUI, physical audio or performance qualification. Retime insertion remains guarded; owned Preserve routes and their lifecycle remain required.'
}
(dest / 'metadata.json').write_text(json.dumps(metadata, indent=2) + '\n')
for name in ['run.py', 'collect.py', 'snapshot.py']:
    (dest / (name + '.txt')).write_bytes((scratch / name).read_bytes())
print(json.dumps({'checks': {k: v.get('tests', v['exit_code']) for k, v in checks.items()}, 'evidence': str(dest)}, indent=2))
