import fcntl
import importlib.util
import json
import os
from pathlib import Path
import subprocess

root = Path('/tmp/deadpan-owner-preparation-20260930')
package = root / 'native-preparation.deadpan'
spec = importlib.util.spec_from_file_location('qualification', root / 'qualify-native-preparation.py')
module = importlib.util.module_from_spec(spec)
spec.loader.exec_module(module)
state = json.loads((root / 'native-preparation-evidence/state.json').read_text())
descriptor = os.open(package / '.writer.lock', os.O_RDONLY | os.O_NOFOLLOW)
try:
    fcntl.flock(descriptor, fcntl.LOCK_EX | fcntl.LOCK_NB)
    fcntl.flock(descriptor, fcntl.LOCK_UN)
finally:
    os.close(descriptor)
assert not (package / '.host.json').exists()
processes = subprocess.check_output(['ps', '-Ao', 'pid,comm'], text=True)
