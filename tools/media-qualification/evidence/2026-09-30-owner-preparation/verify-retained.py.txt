import hashlib
import json
from pathlib import Path
import shutil
import sqlite3

root = Path('/Users/michael/Code/deadpan/tools/media-qualification/evidence/2026-09-30-owner-preparation')
def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
def read(path):
    with sqlite3.connect(path.resolve().as_uri() + '?mode=ro&immutable=1', uri=True) as connection:
        assert connection.execute('PRAGMA quick_check').fetchall() == [('ok',)]
        names = [row[0] for row in connection.execute("SELECT name FROM sqlite_schema WHERE type='table' ORDER BY name")]
        tables = {}
        for name in names:
            assert '"' not in name
            rows = [tuple({'bytes_hex': value.hex()} if isinstance(value, bytes) else value for value in row)
                    for row in connection.execute(f'SELECT * FROM "{name}"')]
            tables[name] = sorted(rows, key=repr)
        return tables

reports = {}
data = {}
for path in sorted((root / 'native').glob('*.sqlite')):
    tables = read(path)
    data[path.name] = tables
    reports[path.name] = {'sha256': digest(path), 'quick_check': 'ok',
                          'table_rows': {name: len(rows) for name, rows in tables.items()}}
assert data['checkpoint.sqlite'] == data['live-backup.sqlite']
assert not any(path.name.endswith(('-shm', '-wal')) for path in root.rglob('*'))
report = {'databases': reports,
          'checkpoint_equals_independent_live_backup': True,
          'comparison': 'Every row of every SQLite table, with exact BLOB bytes and unordered rows sorted for comparison.'}
(root / 'retained-checks.json').write_text(json.dumps(report, indent=2) + '\n')
shutil.copyfile(__file__, root / 'verify-retained.py.txt')
manifest = {str(path.relative_to(root)): {'bytes': path.stat().st_size, 'sha256': digest(path)}
            for path in sorted(root.rglob('*')) if path.is_file() and path.name != 'manifest.json'}
(root / 'manifest.json').write_text(json.dumps(manifest, indent=2) + '\n')
for name, expected in manifest.items():
    path = root / name
    assert path.stat().st_size == expected['bytes'] and digest(path) == expected['sha256']
print(json.dumps({'files': len(manifest), 'bytes': sum(row['bytes'] for row in manifest.values()),
                  'databases': len(data), 'checkpoint_tables_compared': len(data['checkpoint.sqlite'])}))
