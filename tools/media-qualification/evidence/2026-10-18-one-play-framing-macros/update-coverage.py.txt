prepared = []
for name, replacements in changes.items():
    path = root / name
    value = path.read_text()
    if name == 'docs/REQUIREMENTS.md':
        addition = replacements[0][1].removeprefix(replacements[0][0]).strip()
        doubled = addition + ' ' + addition
        if doubled in value:
            assert value.count(doubled) == 1
            value = value.replace(doubled, addition)
    for before, after in replacements:
        if value.count(after) == 1:
            continue
        assert value.count(before) == 1, (name, before, value.count(before))
        value = value.replace(before, after)
    prepared.append((path, value))
