from pathlib import Path
import hashlib
import json
import re
import shutil

root = Path('/Users/michael/Code/deadpan')
scratch = Path('/tmp/deadpan-scoped-hold-duration-2026-10-20')
log = (scratch / 'export.log').read_text()
test = re.search(r'target/release/deps/preview_export-[0-9a-f]+', log).group()
sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
paths = [root / 'target/release/deadpan-cli', root / test]
(scratch / 'export-binaries.json').write_text(json.dumps({str(path.relative_to(root)): sha(path) for path in paths}, indent=2) + '\n')
rows = json.loads((scratch / 'export-results.json').read_text())
assert len(rows) == 2
for index, row in enumerate(rows):
    movie = Path(row['movie'])
    name = 'grow.mp4' if index == 0 else 'shrink.mp4'
    shutil.copy2(movie, scratch / name)
    row['retained_movie'] = name
    row['emitted_sha256'] = sha(movie)
    assert not row['report']['failures']
    assert row['report']['summary']['nonzero_offsets'] == 0
(scratch / 'export-results.json').write_text(json.dumps(rows, indent=2) + '\n')
