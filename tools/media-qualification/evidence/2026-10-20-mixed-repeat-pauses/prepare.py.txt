from pathlib import Path
import hashlib
import json
import shutil

root = Path('/Users/michael/Code/deadpan')
scratch = Path('/tmp/deadpan-mixed-pause-2026-10-20')
binary = scratch / 'bin'
binary.mkdir(exist_ok=True)
for name in ('deadpan-app', 'deadpan-cli'):
    shutil.copy2(root / 'target/release' / name, binary / name)
for name in ('deadpan-media-worker', 'deadpan-track', 'deadpan-transcribe'):
    shutil.copy2(Path('/tmp/deadpan-one-play-framing-2026-10-18/bin') / name, binary / name)
sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
(scratch / 'binaries.json').write_text(json.dumps({
    'sha256': {path.name: sha(path) for path in sorted(binary.iterdir())},
    'app_and_cli_source': '8d226bb5 plus retained source diff',
    'helpers': 'Unchanged private release helpers from one-play framing qualification.',
}, indent=2) + '\n')
sources = sorted(set((scratch / 'tracked-rust-paths.txt').read_text().splitlines() + (scratch / 'new-rust-paths.txt').read_text().splitlines() + ['Cargo.lock', 'rust-toolchain.toml']))
(scratch / 'source-identities.json').write_text(json.dumps({
    'base_revision': '8d226bb5',
    'sha256': {name: sha(root / name) for name in sources},
    'environment': {'hardware': 'Apple M5 Max', 'macOS': '26.5.2 (25F84)', 'Rust': '1.97.1', 'power': 'AC, 80 percent battery; thermal state and caches uncontrolled'},
    'ffmpeg_prefix': '/Users/michael/Library/Developer/Deadpan/ffmpeg-8.0.3-opus-1.6.1-dav1d-1.5.4-qualified-3/prefix',
}, indent=2) + '\n')
print(json.dumps(json.loads((scratch / 'binaries.json').read_text()), indent=2))
