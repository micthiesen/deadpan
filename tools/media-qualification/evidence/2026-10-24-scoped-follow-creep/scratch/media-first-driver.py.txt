def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            h.update(block)
    return h.hexdigest()


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'))


def ratio(value):
    return Fraction(int(value['numerator']), int(value['denominator']))


def round_even(value):
    a, remainder = divmod(value.numerator, value.denominator)
    return a + (2 * remainder > value.denominator or
                (2 * remainder == value.denominator and a % 2 != 0))


Q = 1 << 32


def quantized(value):
    return Fraction(round_even(value * Q), Q)


def scale_at(j):
    first = round_even(Fraction(11 * Q, 10))
    last = round_even(Fraction(11 * Q, 5))
    progress = round_even(Fraction((2 * j + 1) * Q, 72))
    return Fraction(first + round_even(Fraction((last - first) * progress, Q)), Q)

def run(name, *arguments):
    assert sha(cli) == pins['cli_sha256']
    command = [str(cli), *map(str, arguments)]
    started = time.monotonic()
    row = {'name': name, 'argv': command, 'started_unix': time.time(),
           'environment': {key: os.environ.get(key) for key in (
               'DEADPAN_FFMPEG_PREFIX', 'DYLD_LIBRARY_PATH', 'DYLD_FALLBACK_LIBRARY_PATH',
               'DEADPAN_PAUSE_ROUTE_PROFILE')}}
    commands.append(row)
    (args.output / 'commands.json').write_text(json.dumps(commands, indent=2) + '\n')
    with (args.output / f'{name}.stdout').open('xb') as stdout:
        with (args.output / f'{name}.stderr').open('xb') as stderr:
            result = subprocess.run(command, stdout=stdout, stderr=stderr)
    row.update(exit_code=result.returncode, seconds=time.monotonic() - started)
    (args.output / 'commands.json').write_text(json.dumps(commands, indent=2) + '\n')
    result.check_returncode()
    return (args.output / f'{name}.stdout').read_text()


def history(package):
    # Read-only SQLite, including current WAL, never immutable=1 or a writer.
    with sqlite3.connect((package / 'project.sqlite').as_uri() + '?mode=ro', uri=True) as db:
        queries = {
            'revisions': 'SELECT id,parent_id,kind FROM revisions ORDER BY id',
            'history': 'SELECT id,parent_id,revision_id,request,edit FROM history ORDER BY id',
            'redo': 'SELECT position,history_id FROM redo ORDER BY position',
            'state': 'SELECT head_revision,cursor,workflow FROM state ORDER BY singleton',
        }
        values = {name: db.execute(query).fetchall() for name, query in queries.items()}
    return {'counts': {name: len(rows) for name, rows in values.items()},
            'sha256': hashlib.sha256(canonical(values).encode()).hexdigest(),
            'state': values['state'],
            'revision_kinds': {kind: sum(row[2] == kind for row in values['revisions'])
                               for kind in ('initial', 'edit', 'undo', 'redo')}}

def pair(new, old):

    assert nodes[new]['kind']['type'] == nodes[old]['kind']['type']

def normalize(value):
    if isinstance(value, str):
        return aliases.get(value, value)
    if isinstance(value, list):
        return [normalize(v) for v in value]
    if isinstance(value, dict):
        return {k: normalize(v) for k, v in value.items()}
    return value

def source_at(name, position):
    kind = nodes[name]['kind']
    if kind['type'] == 'retime':
        assert kind.get('purpose') == 'partition' and kind['pitch'] == 'preserve'
        mapping = kind['mapping']
        slope = Fraction(mapping['end'] - mapping['start'], kind['duration'])
        assert slope == 1
        return source_at(kind['child'], mapping['start'] + position * slope)
    if kind['type'] == 'sequence':
        for child in kind['children']:
            length = duration(child)
            if position < length:
                return source_at(child, position)
            position -= length
        raise AssertionError('Source point outside Sequence')
    assert kind['type'] == 'source'
    source = kind['source']
    video = source['video']
    assert video['type'] == 'stream' and video['asset'] == target['asset']
    assert video['span']['start']['time_base'] == video['span']['end']['time_base'] == clock
    mapping = source['video_mapping']
    if mapping['type'] == 'fit_beat':
        placement, frames = Fraction(0), Fraction(source['duration'])
    else:
        assert mapping['type'] in ('duration', 'placement', 'selected_placement')
        placement = ratio(mapping['start']) if 'start' in mapping else Fraction(0)
        frames = ratio(mapping['frames'])
    span = video['span']
    ticks = span['start']['ticks'] + (position - placement) * (
        span['end']['ticks'] - span['start']['ticks']) / frames
    return name, ticks


def duration(name):
    kind = nodes[name]['kind']
    if kind['type'] == 'source':
        return kind['source']['duration']
    if kind['type'] == 'retime':
        return kind['duration']
    if kind['type'] == 'sequence':
        return sum(duration(c) for c in kind['children'])
    if kind['type'] == 'repeat':
        count = sum(r['count'] for r in kind['iterations']['runs'])
        return count * duration(kind['child']) + (count - 1) * kind.get('gap', {}).get('duration', 0)
    raise AssertionError(kind['type'])
