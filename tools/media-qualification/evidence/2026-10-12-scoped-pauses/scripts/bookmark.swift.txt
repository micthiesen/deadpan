import Foundation
let fm = FileManager.default
let root = fm.temporaryDirectory.appendingPathComponent("deadpan-bookmark-\(UUID().uuidString)")
try fm.createDirectory(at: root, withIntermediateDirectories: true)
defer { try? fm.removeItem(at: root) }
let original = root.appendingPathComponent("clip.mov")
let moved = root.appendingPathComponent("moved.mov")
try Data("bookmark diagnostic".utf8).write(to: original)
do {
    let bookmark = try original.bookmarkData(options: [], includingResourceValuesForKeys: nil, relativeTo: nil)
    print("bookmark bytes: \(bookmark.count)")
    try fm.moveItem(at: original, to: moved)
    var stale = false
    let resolved = try URL(resolvingBookmarkData: bookmark, options: [.withoutUI, .withoutMounting], relativeTo: nil, bookmarkDataIsStale: &stale)
    print("resolved moved file: \(resolved.standardizedFileURL == moved.standardizedFileURL); stale: \(stale)")
} catch {
    let error = error as NSError
    print("bookmark failure: \(error.domain) \(error.code): \(error.localizedDescription)")
    exit(1)
}
