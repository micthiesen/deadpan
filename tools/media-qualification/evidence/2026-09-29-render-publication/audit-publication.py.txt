"""Independently inspect final names, byte identities and captured report bindings."""
import hashlib
import json
from pathlib import Path
import stat
import sqlite3

ROOT = Path(__file__).resolve().parent


def digest(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


run = json.loads((ROOT / 'metal-publication-report.json').read_text())
cases = run['encoded']['cases']
assert len(cases) == 7
observations = []
for case in cases:
    assert case['publication_status'] == 'published', case['name']
    receipt = case['publication']
    movie, report = Path(receipt['movie']), Path(receipt['report'])
    candidate = Path(case['encoded_candidate_path'])
    assert str(movie) == case['path']
    for path, size, checksum in [
        (movie, receipt['movie_bytes'], receipt['movie_sha256']),
        (report, receipt['report_bytes'], receipt['report_sha256']),
    ]:
        metadata = path.lstat()
        assert stat.S_ISREG(metadata.st_mode) and metadata.st_nlink == 1
        assert stat.S_IMODE(metadata.st_mode) == 0o600
        assert metadata.st_size == size and digest(path) == checksum
    assert digest(candidate) == receipt['movie_sha256']
    local = json.loads(report.read_text())
    assert local['schema_version'] == 1
    assert local['publication_id'] == receipt['publication_id']
    assert local['movie_filename'] == movie.name
    assert local['report_filename'] == report.name
    assert local['destination_readback'] == {
        'bytes': receipt['movie_bytes'], 'sha256': receipt['movie_sha256']}
    provenance = local['provenance']
    contract = case['contract']
    for key in ['project_id', 'revision_id', 'range']:
        assert provenance[key] == contract[key], (case['name'], key)
    assert provenance['document_sha256'] == case['manifest']['document_sha256']
    assert provenance['encoded_manifest'] == case['manifest']
    assert provenance['verification'] == case['verification']
    assert provenance['encoder_selection'] == 'explicit_engineering_choice'
    assert provenance['catalog_scope'] == 'committed_catalog_superset'
    qualified = [entry for entry in provenance['catalog']
                 if entry['qualification']['status'] == 'qualified']
    generated = case['name'] == 'generated'
    # The accepted Generated fixture intentionally has no Original receipts.
    assert bool(qualified) == (not generated), case['name']
    package = (Path('/tmp/deadpan-generated-pictures-smyys035/fixture/accepted.deadpan')
               if generated else ROOT / 'metal-publication-files' / {
                   'odd': 'odd-canvas.deadpan', 'marker': 'marker.deadpan'
               }.get(case['name'], 'picture.deadpan'))
    with sqlite3.connect((package / 'project.sqlite').as_uri() + '?mode=ro', uri=True) as database:
        historical = database.execute('SELECT document FROM revisions WHERE id = ?',
                                      (provenance['revision_id'],)).fetchone()[0]
        compact = json.dumps(json.loads(historical), ensure_ascii=False, separators=(',', ':'))
        assert hashlib.sha256(compact.encode()).hexdigest() == provenance['document_sha256']
        source_receipts = {row[0]: (json.loads(row[1]), json.loads(row[2]))
                           for row in database.execute('SELECT id, original_ref, snapshot FROM source_qualifications')}
    for entry in qualified:
        evidence = entry['qualification']
        assert len(evidence['source_sha256']) == 64
        original, snapshot = source_receipts[evidence['receipt']]
        assert original == evidence['original']
        assert bytes(snapshot['content']['sha256']).hex() == evidence['source_sha256']
        content = original['content']
        source = package / 'Media' / 'Originals' / (content['algorithm'] + '-' + content['digest'])
        assert source.stat().st_size == original['byte_length']
        assert digest(source) == evidence['source_sha256']
    assert receipt['contains_generated_pictures'] == generated
    assert bool(provenance['generated_intervals']) == generated
    assert bool(provenance['generated_artifacts']) == generated
    if generated:
        assert provenance['generated_intervals'] == [{
            'artifact_index': 0, 'project_range': {'start': 0, 'end': 30},
            'output_range': [0, 30]}]
    assert case['publication_progress'] == [
        'capturing_provenance', 'copying_destination', 'checking_destination',
        'writing_report', 'ready_to_publish']
    observations.append({
        'case': case['name'], 'movie': str(movie), 'report': str(report),
        'movie_sha256': receipt['movie_sha256'],
        'report_sha256': receipt['report_sha256'],
        'movie_bytes': receipt['movie_bytes'], 'report_bytes': receipt['report_bytes'],
        'revision': provenance['revision_id'], 'qualified_catalog_entries': len(qualified),
        'generated_intervals': provenance['generated_intervals'],
    })
assert not list((ROOT / 'metal-publication-files').glob('*.partial'))
result = {'passed': True, 'cases': observations,
          'scope': 'Final files read independently; exact movie/report hashes, private singleton metadata, provenance bindings and effective generated intervals.'}
(ROOT / 'publication-audit.json').write_text(json.dumps(result, indent=2) + '\n')
print(json.dumps(result, indent=2))
