"""Run Cargo-recorded CLI/example artifacts and retain exact input identities."""
import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys

root = Path(__file__).resolve().parent
name, build_name = sys.argv[1:]
records = []
for line in (root / (build_name + '.log')).read_text().splitlines():
    try:
        records.append(json.loads(line))
    except json.JSONDecodeError:
        pass
artifacts = {}
for target, kind in [('deadpan-cli', 'bin'), ('qualify_project_picture', 'example')]:
    matches = [row for row in records if row.get('reason') == 'compiler-artifact'
               and row.get('target', {}).get('name') == target
               and row.get('target', {}).get('kind') == [kind] and row.get('executable')]
    assert len(matches) == 1, (target, len(matches))
    row = matches[0]
    original = Path(row['executable']).resolve()
    binary_dir = root / (name + '-bin')
    binary_dir.mkdir(exist_ok=True)
    path = binary_dir / target
    shutil.copy2(original, path)
    artifacts[target] = {'cargo_artifact': row, 'original': str(original), 'path': str(path), 'sha256': hashlib.sha256(path.read_bytes()).hexdigest()}
marker = Path('/tmp/deadpan-native-encode-i8gk62qm/normal-final/hardware-none-60.mp4')
binary = Path(artifacts['qualify_project_picture']['path'])
environment = os.environ | {'PATH': str(binary.parent) + os.pathsep + os.environ['PATH']}
assert Path(shutil.which(binary.name, path=environment['PATH'])).resolve() == binary
command = [binary.name, str(root / (name + '-report.json')), str(root / (name + '-files')),
           '/tmp/deadpan-generated-pictures-smyys035/fixture/accepted.deadpan',
           artifacts['deadpan-cli']['path'], '--encoded', str(marker)]
receipt = {'artifacts': artifacts, 'command': command, 'cwd': str(Path.cwd()), 'timeout_seconds': 660,
           'marker': {'path': str(marker), 'sha256': hashlib.sha256(marker.read_bytes()).hexdigest()}}
path = root / (name + '-artifacts.json')
with path.open('x') as file:
    json.dump(receipt, file, indent=2)
    file.write('\n')
try:
    result = subprocess.run(command, env=environment, timeout=660)
    receipt['exit_code'] = result.returncode
except subprocess.TimeoutExpired:
    receipt.update(timed_out=True, exit_code=124)
for artifact in artifacts.values():
    artifact['sha256_after'] = hashlib.sha256(Path(artifact['path']).read_bytes()).hexdigest()
    artifact['unchanged'] = artifact['sha256_after'] == artifact['sha256']
path.write_text(json.dumps(receipt, indent=2) + '\n')
assert all(artifact['unchanged'] for artifact in artifacts.values())
sys.exit(receipt['exit_code'])
