from pathlib import Path
import hashlib,json,subprocess,time
root=Path(__file__).resolve().parent
cli=Path('/tmp/deadpan-inference-pause-final-bin-2026-10-17/deadpan-cli')
def sha(p):
 with p.open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def save(p,v):p.write_text(json.dumps(v,indent=2)+'\n')
record={'executable':str(cli),'executable_sha256':sha(cli),'identity':'Retained qualified standalone CLI from 2026-10-17 inference scheduling run; not rebuilt for this preparation','runs':[]}
for name,fixture in [('1080p','gen-1080p60.mp4'),('4k','gen-4k30.mp4')]:
 source=Path('/tmp/dp-perf-after-1/fixtures')/fixture;package=root/(name+'.deadpan')
 assert not package.exists(),package
 original_hash=sha(source)
 commands=[]
 for action,args in [('create',['project','create-original',str(package),str(source)]),('dump',['project','dump',str(package),'--json']),('validate',['project','validate',str(package),'--quick'])]:
  command=[str(cli),*args];start=time.monotonic();p=subprocess.run(command,capture_output=True,text=True)
  (root/(name+'-'+action+'.json')).write_text(p.stdout);(root/(name+'-'+action+'.stderr')).write_text(p.stderr)
  commands.append({'command':command,'exit_code':p.returncode,'seconds':time.monotonic()-start})
  if p.returncode:raise RuntimeError((name,action,p.stderr))
 assert sha(source)==original_hash,'Original changed'
 record['runs'].append({'name':name,'project':str(package),'fixture':str(source),'fixture_bytes':source.stat().st_size,'fixture_sha256':original_hash,'commands':commands})
 save(root/'fixture-preparation.json',record)
print(json.dumps(record,indent=2))
