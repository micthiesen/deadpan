from pathlib import Path
import json, os, subprocess, time

root = Path('/tmp/deadpan-extension-longer-20261008')
repo = Path('/Users/michael/Code/deadpan')
original = json.loads((root/'final-checks-results.json').read_text())
assert original[-1]['name'] == 'gate', 'Wait for the original gate to finish'
env = dict(os.environ, DEADPAN_FFMPEG_PREFIX='/Users/michael/Library/Developer/Deadpan/ffmpeg-8.0.3/prefix')
checks = [
    ('correction-fmt', ['cargo','fmt','--all','--check']),
    ('correction-clippy', ['cargo','clippy','-p','deadpan-app','--all-targets','--locked','--','-D','warnings']),
    ('correction-ui-clippy', ['cargo','clippy','-p','deadpan-app','--all-targets','--features','ui-harness','--locked','--','-D','warnings']),
    ('correction-tests', ['cargo','nextest','run','-p','deadpan-app','--locked','-E','test(project::tests::generation)','--no-fail-fast']),
    ('correction-ui-tests', ['cargo','nextest','run','-p','deadpan-app','--features','ui-harness','--locked','-E','test(project::tests::generation)','--no-fail-fast']),
    ('doctests', ['cargo','test','--workspace','--features','deadpan-cli/synthetic-worker','--locked','--doc','--no-fail-fast']),
]
results=[]
for name, command in checks:
    start=time.monotonic()
    with (root/(name+'.log')).open('x') as output:
        process=subprocess.run(command,cwd=repo,env=env,stdout=output,stderr=subprocess.STDOUT)
    result={'name':name,'command':command,'exit_code':process.returncode,'seconds':time.monotonic()-start}
    results.append(result)
    (root/'correction-results.json').write_text(json.dumps(results,indent=2)+'\n')
    print(json.dumps(result),flush=True)
    if process.returncode:
        print((root/(name+'.log')).read_text()[-6000:],flush=True)
        raise SystemExit(process.returncode)
