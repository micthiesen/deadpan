from pathlib import Path
import json, os, subprocess, time

root = Path('/tmp/deadpan-extension-longer-20261008')
repo = Path('/Users/michael/Code/deadpan')
env = dict(os.environ, DEADPAN_FFMPEG_PREFIX='/Users/michael/Library/Developer/Deadpan/ffmpeg-8.0.3/prefix')
checks = [
    ('capture-build', ['cargo','build','--locked','-p','deadpan-cli','--bin','deadpan-cli','--example','prepare_extension_context']),
    ('model-tools-build', ['cargo','build','--locked','-p','deadpan-jobs','--example','qualify_model_worker','-p','deadpan-models','--example','retain_extension_conditioning','--example','qualify_extension_media']),
    ('capture-tests', ['cargo','nextest','run','--locked','-p','deadpan-cli','--lib','-E','test(generation::conditioning::extension) | test(picture::tests::extension)','--no-fail-fast']),
]
results = []
for name, command in checks:
    start = time.monotonic()
    with (root/(name+'.log')).open('x') as output:
        process = subprocess.run(command,cwd=repo,env=env,stdout=output,stderr=subprocess.STDOUT)
    results.append({'name':name,'command':command,'exit_code':process.returncode,'seconds':time.monotonic()-start})
    (root/'checks-results.json').write_text(json.dumps(results,indent=2)+'\n')
    print(name,process.returncode,flush=True)
    if process.returncode:
        print((root/(name+'.log')).read_text()[-6000:],flush=True)
        raise SystemExit(process.returncode)
