def save(path, value):
    with path.open('x') as f:
        json.dump(value, f, indent=2)
        f.write('\n')

def sha(path):
    return hashlib.file_digest(path.open('rb'), 'sha256').hexdigest()

def ratio(value):
    return dict(numerator=str(value.numerator), denominator=str(value.denominator))
