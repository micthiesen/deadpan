import json
import os
from pathlib import Path
import subprocess
import time

root=Path('/tmp/deadpan-resume-20261006/extension-worker/checks-final')
root.mkdir()
env=os.environ.copy()
env['DEADPAN_FFMPEG_PREFIX']='/Users/michael/Library/Developer/Deadpan/ffmpeg-8.0.3/prefix'
steps=[
 ('fmt',['cargo','fmt','--all']),
 ('tests',['cargo','test','--locked','-p','deadpan-models','--lib','conditioning::extension']),
 ('protocol-fuzz',['cargo','test','--locked','-p','deadpan-jobs','--test','protocol','adversarial_generation_frames']),
 ('clippy',['cargo','clippy','--locked','-p','deadpan-models','-p','deadpan-jobs','-p','xtask','--all-targets','--','-D','warnings']),
 ('chaos',['cargo','xtask','chaos','--only','jobs-generation,model-packs','--minutes','0.5','--seed','20261008','--output',str(root/'chaos')]),
 ('diff',['git','diff','--check'])]
results=[]
for name,args in steps:
    start=time.monotonic()
    with (root/(name+'.log')).open('wb') as f:
        result=subprocess.run(args,cwd='/Users/michael/Code/deadpan',env=env,stdout=f,stderr=subprocess.STDOUT)
    results.append(dict(name=name,command=args,exit=result.returncode,seconds=round(time.monotonic()-start,3)))
    (root/'results.json').write_text(json.dumps(results,indent=2)+'\n')
    print(json.dumps(results[-1]),flush=True)
    if result.returncode:
        print((root/(name+'.log')).read_text()[-15000:],flush=True)
        raise SystemExit(result.returncode)
