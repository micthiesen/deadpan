#!/usr/bin/env python3
"""Sequential final builds and real-path checks, stopping at the first failure."""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import time

root = Path('/tmp/deadpan-resume-20261006')
repo = Path('/Users/michael/Code/deadpan')
env = dict(os.environ)
env['DEADPAN_FFMPEG_PREFIX'] = '/Users/michael/Library/Developer/Deadpan/ffmpeg-8.0.3/prefix'
env['DEADPAN_REQUIRE_SYNTHETIC_WORKER'] = '1'
reports = []

def run(name, command, extra=None):
    started = time.monotonic()
    environment = dict(env)
    environment.update(extra or {})
    load = subprocess.check_output(['uptime'], text=True).strip()
    with (root / f'boundary-release-{name}.log').open('w') as log:
        result = subprocess.run([str(item) for item in command], cwd=repo,
                                env=environment, stdout=log, stderr=subprocess.STDOUT)
    report = {'name': name, 'command': [str(item) for item in command],
              'environment_overrides': extra or {}, 'exit': result.returncode,
              'seconds': round(time.monotonic() - started, 3), 'load_before': load}
    reports.append(report)
    (root / 'boundary-release-checks.json').write_text(json.dumps(reports, indent=2) + '\n')
    print(json.dumps(report), flush=True)
    if result.returncode:
        raise SystemExit(result.returncode)

paths = subprocess.check_output(['git', 'ls-files', '-m', '-o', '--exclude-standard', '-z'], cwd=repo).split(b'\0')
source = {}
for path in sorted(set(paths)):
    if not path:
        continue
    path = path.decode()
    if path.startswith(('crates/', 'native/')):
        file = repo / path
        source[path] = hashlib.sha256(file.read_bytes()).hexdigest() if file.is_file() else None
(root / 'boundary-final-source.json').write_text(json.dumps({
    'base_commit': subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=repo, text=True).strip(),
    'changed_source_sha256': source,
}, indent=2) + '\n')

build = root / 'boundary-final-release-build'
run('build', ['python3', root / 'build-insertion-snapshots.py', '--release', '--output', build])
app = build / 'bin/deadpan-app'
helpers = {'DEADPAN_MEDIA_WORKER': str(build / 'bin/deadpan-media-worker')}
run('generated-exports', ['cargo', 'test', '--release', '--locked', '-p', 'deadpan-cli',
    '--features', 'synthetic-worker', '--test', 'preview_export',
    'every_recipe_export_matches_its_committed_preview', '--', '--nocapture'], {
        **helpers, 'DEADPAN_PREVIEW_EXPORT_ONLY': 'generated-pause,generated-repeat,generated-reframe,generated-prefix',
        'DEADPAN_PREVIEW_EXPORT_RESULTS': str(root / 'boundary-generated-exports.json')})
run('compaction', ['cargo', 'nextest', 'run', '--release', '--locked', '-p', 'deadpan-store',
    '--test', 'generation_bundles', '-E',
    'test(compacted_terminal_preparation_still_proves_birth_and_supplies_redo_controls)'])
for scenario in ['ai-boundaries', 'ai-replacements', 'ai-scoped', 'ai-variants', 'ai-insertion']:
    run('visual-' + scenario, [app, '--ui-check', '--mode', 'visual', '--scenario', scenario,
        '--output', root / ('boundary-final-visual-' + scenario)], helpers)
for scenario in ['ai-boundaries', 'rapid-input', 'edit-latency', 'large-project']:
    run('performance-' + scenario, [app, '--ui-check', '--mode', 'performance', '--scenario', scenario,
        '--output', root / ('boundary-performance-' + scenario)], helpers)
bundle = root / 'boundary-final-bundle'
run('bundle', ['cargo', 'xtask', 'bundle', '--output', bundle, '--allow-dirty'])
run('bundle-verify', ['cargo', 'xtask', 'bundle-verify', bundle / 'Deadpan.app'])
run('packaged-export', ['python3', root / 'verify-boundary-generation.py', '--bin-dir',
    bundle / 'Deadpan.app/Contents/MacOS', '--project',
    root / 'boundary-real-generation-1/boundary-generation.deadpan', '--expected-report',
    root / 'boundary-real-generation-1/summary.json', '--output', root / 'boundary-packaged-generation-1'])
